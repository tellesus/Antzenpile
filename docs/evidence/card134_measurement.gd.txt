func go() -> void:
	for scenario: String in ["backyard_slice", "garden_edge", "roadside"]:
		for seed_value: int in [482817, 71, 3030]:
			var root = Root.new()
			root.simulation = Controller.new(seed_value, scenario)
			var row: Dictionary = {"scenario":scenario,"seed":seed_value,"first_report":{},"first_delivery":{},"first_emergence":-1,"food_short_samples":0,"actions":[],"avoided":[],"watch_attempted":[]}
			root.set_exploration(5)
			root.set_brood_intent("grow")
			for sample: int in 240:
				observe_and_act(root,row)
				root.simulation.advance(5.0)
			var inside: Dictionary = root.inward_status("home")
			row.final = {"time":inside.time,"workers":inside.workers_total,"emerged":inside.brood_matured_total,"brood_losses":inside.brood_losses,"stores":inside.resources,"nursery":inside.nursery_state,"food_exchange":inside.food_exchange_state,"brood_waiting":inside.brood_production}

			print("[REVIEW ECONOMY] ",scenario," ",seed_value," emerged=",row.final.emerged," short_samples=",row.food_short_samples," first=",row.first_report," exact=",row.save_exact)

			var root = Root.new(); root.simulation = Controller.new()
			if not root.simulation.restore_snapshot(saved): failures.append("combat restore")
			var before: Dictionary = root.inward_status("home")
			var game: SimulationController = root.simulation

			if choice == "bypass": accepted = game.journey_response.investigate_approach("route_1")

				game.journey_response.set_goal("route_1","hunt" if choice.begins_with("hunt") else "clear")
				accepted = game.journey_response.set_force("route_1",24 if choice.ends_with("24") else 12)

func record(row: Dictionary, time: float, label: String, result: Dictionary) -> void:
	if result.get("accepted",false): row.actions.append({"time":time,"action":label})

func observe_and_act(root: Node, row: Dictionary) -> void:
	var inside: Dictionary = root.inward_status("home")
	var outside: Dictionary = root.outward_status("home")
	var signals: Array[Dictionary] = root.sensory_snapshot("home")
	if inside.brood_matured_total > 0 and row.first_emergence < 0: row.first_emergence=inside.time
	if not ColonyPressure.food_shortages(inside).is_empty(): row.food_short_samples += 1
	for signal_data: Dictionary in signals:
		if not row.first_report.has(signal_data.category): row.first_report[signal_data.category]=inside.time
	for route: Dictionary in outside.trails:
		if route.delivered_total > 0:
			for signal_data: Dictionary in signals:
				if signal_data.source_knowledge_id==route.destination_knowledge_id and not row.first_delivery.has(signal_data.category): row.first_delivery[signal_data.category]=inside.time
		if route.journey_alarm and route.id not in row.avoided:
			row.avoided.append(route.id);record(row,inside.time,"recall alarm "+route.id,root.set_trail_target(route.id,0))
		elif route.status == "depleted" and not route.resume_on_report and route.id not in row.watch_attempted:
			row.watch_attempted.append(route.id);record(row,inside.time,"watch "+route.id,root.toggle_investigation_priority(route.destination_knowledge_id))
	for category: String in ["carbohydrate","protein","water"]:
		var entries: Array[Dictionary] = Memories.entries(signals,outside,category)
		var active: bool = false
		for entry: Dictionary in entries:
			if entry.desired_workers > 0 and entry.route_status == "active" and not entry.danger: active=true
		if active: continue
		entries.sort_custom(func(a: Dictionary,b: Dictionary)->bool:return a.age<b.age)
		for entry: Dictionary in entries:
			if entry.danger or entry.route_id in row.avoided or entry.state == "Reported empty": continue
			if entry.route_status == "depleted":
				record(row,inside.time,"retry "+entry.route_id,root.recheck_trail(entry.route_id))
			elif entry.route_status in ["none","inactive"]:
				record(row,inside.time,"gather "+category,root.create_trail_for(entry.knowledge_id))
			break
	if inside.food_exchange_state == "primitive": record(row,inside.time,"develop exchange",root.start_food_exchange())
	if inside.food_exchange_state == "developed" and inside.nursery_state == "primitive": record(row,inside.time,"develop nursery",root.start_nursery_development())
	if inside.midden.revealed and inside.midden.larval_rate < 1.0 and inside.midden.cleaners == 0: record(row,inside.time,"clean 2",root.set_sanitation_workers(2))
	if inside.nursery_state == "developed" and inside.humidity.larval_rate < 1.0 and inside.humidity.carers == 0: record(row,inside.time,"climate 1",root.set_humidity_workers(1))
	if inside.guest.get("reported_losses",0)>0 and inside.guest.observation != "purged" and not inside.guest.rejection_active: record(row,inside.time,"reject guest",root.set_guest_rejection(true))
